    }

    public function create_location_table() {
        global $wpdb;
        $table_name = $wpdb->prefix . 'enhanced_user_locations';
        
        $charset_collate = $wpdb->get_charset_collate();
        
        $sql = "CREATE TABLE $table_name (
            id mediumint(9) NOT NULL AUTO_INCREMENT,
            user_id bigint(20) NOT NULL,
            latitude decimal(10,7) NOT NULL,
            longitude decimal(10,7) NOT NULL,
            address text NOT NULL,
            full_name varchar(255),
            telephone varchar(20),
            email varchar(100),
            profile_picture varchar(255),
            additional_info text,
            PRIMARY KEY  (id)
        ) $charset_collate;";
        
        require_once(ABSPATH . 'wp-admin/includes/upgrade.php');
        dbDelta($sql);
    }

    public function save_user_location() {
        // Verify nonce for security
        check_ajax_referer('save_user_location_action', 'user_location_nonce');
        
        // Check user capabilities
        if (!current_user_can('manage_options')) {
            wp_send_json_error('Unauthorized');
        }
        
        // Sanitize and validate input
        $user_id = intval($_POST['user_id']);
        $latitude = floatval($_POST['latitude']);
        $longitude = floatval($_POST['longitude']);
        $address = sanitize_text_field($_POST['address']);
        $full_name = sanitize_text_field($_POST['full_name']);
        $telephone = sanitize_text_field($_POST['telephone']);
        $email = sanitize_email($_POST['email']);
        $additional_info = sanitize_textarea_field($_POST['additional_info']);
        
        // Handle file upload
        $profile_picture = '';
        if (!empty($_FILES['profile_picture']['name'])) {
            $upload = wp_handle_upload($_FILES['profile_picture'], [
                'test_form' => false
            ]);
            
            if (!isset($upload['error'])) {
                $profile_picture = $upload['url'];
            }
        }
        
        global $wpdb;
        $table_name = $wpdb->prefix . 'enhanced_user_locations';
        
        // Check if user location already exists
        $existing = $wpdb->get_row($wpdb->prepare(
            "SELECT id FROM $table_name WHERE user_id = %d",
            $user_id
        ));
        
        $data = [
            'user_id' => $user_id,
            'latitude' => $latitude,
            'longitude' => $longitude,
            'address' => $address,
            'full_name' => $full_name,
            'telephone' => $telephone,
            'email' => $email,
            'profile_picture' => $profile_picture,
            'additional_info' => $additional_info
        ];
        
        if ($existing) {
            // Update existing record
            $result = $wpdb->update(
                $table_name, 
                $data, 
                ['user_id' => $user_id]
            );
        } else {
            // Insert new record
            $result = $wpdb->insert($table_name, $data);
        }
        
        if ($result !== false) {
            wp_send_json_success('User location saved');
        } else {
            wp_send_json_error('Database error');
        }
    }

    public function enqueue_frontend_scripts() {
        // Enqueue Google Maps and custom JS

    }

    public function find_nearby_users() {
        global $wpdb;
        $table_name = $wpdb->prefix . 'enhanced_user_locations';
        
        $latitude = floatval($_POST['latitude']);
        $longitude = floatval($_POST['longitude']);
        $radius = 10; // Default radius in miles
        
        $nearby_users = $wpdb->get_results($wpdb->prepare(
            "SELECT user_id, full_name, telephone, email, profile_picture, additional_info,
            (6371 * acos(cos(radians(%f)) * cos(radians(latitude)) * 
            cos(radians(longitude) - radians(%f)) + 
            sin(radians(%f)) * sin(radians(latitude)))) AS distance 
            FROM $table_name 
            HAVING distance < %f 
            ORDER BY distance 
            LIMIT 10",
            $latitude, $longitude, $latitude, $radius
        ));
        
        wp_send_json_success($nearby_users);
    }
